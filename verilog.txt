`timescale 1ns / 1ps

module priority_encoder_4x2 (
    input  wire [3:0] in,
    output reg  [1:0] out,
    output reg        valid
);

    always @(*) begin
        // Default values to prevent latches
        valid = 1'b1;
        out   = 2'b00;

        casez (in)
            4'b1???: out = 2'b11; // Highest priority (Bit 3)
            4'b01??: out = 2'b10; // Priority to Bit 2
            4'b001?: out = 2'b01; // Priority to Bit 1
            4'b0001: out = 2'b00; // Lowest priority (Bit 0)
            default: begin
                // Case where in == 4'b0000
                out   = 2'b00;
                valid = 1'b0;     
            end
        endcase
    end

endmodule
